import {
  findBrandStoreMapping,
  saveBrandStoreMapping,
} from "./brandStoreMappings";
type SerpApiResult = {
  ads_results?: {
    site?: string;
    title?: string;
    sub_title?: string;
    description?: string;
  }[];

  web_results?: {
    title?: string;
    link?: string;
    snippet?: string;
  }[];

  shopping_results?: {
    title?: string;
    link?: string;
    stores?: string;
    price?: number | string;
    rating?: number | string;
    reviews?: number | string;
  }[];
};

type SerpApiResponse =
  SerpApiResult & {
    error?: string;
  };

type CanonicalCandidate = {
  productId: string;
  brandSite: string;
  canonicalUrl: string;
  title: string;
};

function normalizeBrandSite(
  value: unknown,
) {
  const text =
    String(value ?? "")
      .trim();

  const match =
    text.match(
      /(?:https?:\/\/)?brand\.naver\.com\/([a-zA-Z0-9_-]+)/i,
    );

  if (!match) {
    return "";
  }

  return (
    "https://brand.naver.com/" +
    match[1]
  );
}

function normalizeCanonicalProductUrl(
  value: unknown,
) {
  const text =
    String(value ?? "")
      .trim();

  const match =
    text.match(
      /https?:\/\/brand\.naver\.com\/([a-zA-Z0-9_-]+)\/products\/(\d+)/i,
    );

  if (!match) {
    return null;
  }

  const brandSite =
    `https://brand.naver.com/${match[1]}`;

  const productId =
    match[2];

  return {
    productId,
    brandSite,
    canonicalUrl:
      `${brandSite}/products/${productId}`,
  };
}

async function searchNaver(
  query: string,
): Promise<SerpApiResponse> {
  const apiKey =
    process.env.SERPAPI_API_KEY;

  if (!apiKey) {
    throw new Error(
      "SERPAPI_API_KEY가 설정되지 않았습니다.",
    );
  }

  const params =
    new URLSearchParams({
      engine: "naver",
      query,
      where: "nexearch",
      output: "json",
      api_key: apiKey,
    });

  const response =
    await fetch(
      "https://serpapi.com/search?" +
        params.toString(),
      {
        cache: "no-store",
      },
    );

  const data =
    (await response.json()) as
      SerpApiResponse;

  if (
    !response.ok ||
    data.error
  ) {
    throw new Error(
      data.error ??
        `SerpApi 검색 실패 (${response.status})`,
    );
  }

  return data;
}

async function searchGoogle(
  query: string,
) {
  const apiKey =
    process.env.SERPAPI_API_KEY;

  if (!apiKey) {
    throw new Error(
      "SERPAPI_API_KEY가 설정되지 않았습니다.",
    );
  }

  const params =
    new URLSearchParams({
      engine: "google",
      q: query,
      api_key: apiKey,
      num: "10",
    });

  const response =
    await fetch(
      "https://serpapi.com/search?" +
        params.toString(),
      {
        cache: "no-store",
      },
    );

  const data =
    (await response.json()) as {
      error?: string;
      organic_results?: {
        title?: string;
        link?: string;
      }[];
    };

  if (
    !response.ok ||
    data.error
  ) {
    /*
      Google이 결과 없음 메시지를 error로 돌려주는 경우는
      resolver 전체 실패로 처리하지 않는다.
    */
    if (
      typeof data.error ===
        "string" &&
      data.error
        .toLowerCase()
        .includes(
          "hasn't returned any results",
        )
    ) {
      return [];
    }

    throw new Error(
      data.error ??
        `Google SerpApi 검색 실패 (${response.status})`,
    );
  }

  const candidates =
    new Map<
      string,
      CanonicalCandidate
    >();

  for (
    const result of
    data.organic_results ??
      []
  ) {
    const normalized =
      normalizeCanonicalProductUrl(
        result.link,
      );

    if (!normalized) {
      continue;
    }

    candidates.set(
      normalized.canonicalUrl,
      {
        ...normalized,

        title:
          String(
            result.title ??
              "",
          ),
      },
    );
  }

  /*
    구조화 결과 밖에 URL이 있을 수도 있으므로
    전체 응답에서도 brand 상품 URL을 확인한다.
  */
  const serialized =
    JSON.stringify(data);

  const matches =
    serialized.match(
      /https?:\/\/brand\.naver\.com\/[a-zA-Z0-9_-]+\/products\/\d+/gi,
    ) ?? [];

  for (
    const match of matches
  ) {
    const normalized =
      normalizeCanonicalProductUrl(
        match,
      );

    if (
      normalized &&
      !candidates.has(
        normalized.canonicalUrl,
      )
    ) {
      candidates.set(
        normalized.canonicalUrl,
        {
          ...normalized,
          title: "",
        },
      );
    }
  }

  return [
    ...candidates.values(),
  ];
}

function extractBrandSites(
  data: SerpApiResult,
) {
  const sites =
    new Set<string>();

  const add = (
    value: unknown,
  ) => {
    const site =
      normalizeBrandSite(
        value,
      );

    if (site) {
      sites.add(site);
    }
  };

  for (
    const ad of
    data.ads_results ?? []
  ) {
    add(ad.site);
  }

  for (
    const result of
    data.web_results ?? []
  ) {
    add(result.link);
  }

  for (
    const result of
    data.shopping_results ?? []
  ) {
    add(result.link);
  }

  const serialized =
    JSON.stringify(data);

  const matches =
    serialized.match(
      /brand\.naver\.com\/[a-zA-Z0-9_-]+/gi,
    ) ?? [];

  for (
    const match of
    matches
  ) {
    add(match);
  }

  return [...sites];
}

function extractCanonicalCandidates(
  data: SerpApiResult,
) {
  const candidates =
    new Map<
      string,
      CanonicalCandidate
    >();

  const add = (
    value: unknown,
    title = "",
  ) => {
    const normalized =
      normalizeCanonicalProductUrl(
        value,
      );

    if (!normalized) {
      return;
    }

    const key =
      normalized.canonicalUrl;

    if (
      !candidates.has(key)
    ) {
      candidates.set(
        key,
        {
          ...normalized,
          title,
        },
      );
    }
  };

  for (
    const result of
    data.web_results ?? []
  ) {
    add(
      result.link,
      String(
        result.title ?? "",
      ),
    );
  }

  for (
    const result of
    data.shopping_results ?? []
  ) {
    add(
      result.link,
      String(
        result.title ?? "",
      ),
    );
  }

  /*
    구조화 필드 밖에 공식 상품 URL이
    들어오는 경우도 대응한다.
  */
  const serialized =
    JSON.stringify(data);

  const matches =
    serialized.match(
      /https?:\/\/brand\.naver\.com\/[a-zA-Z0-9_-]+\/products\/\d+/gi,
    ) ?? [];

  for (
    const match of
    matches
  ) {
    add(match);
  }

  return [
    ...candidates.values(),
  ];
}

function extractBrandName(
  data: SerpApiResult,
) {
  for (
    const result of
    data.web_results ?? []
  ) {
    const combined =
      [
        result.title,
        result.snippet,
      ]
        .filter(Boolean)
        .join(" ");

    const match =
      combined.match(
        /브랜드\s*[:：]\s*([가-힣A-Za-z0-9._+-]+)/,
      );

    if (match?.[1]) {
      return match[1].trim();
    }
  }

  return "";
}

function getProductId(
  productUrl: string,
) {
  /*
    실제 네이버 스토어 상품 URL에 포함된
    productId만 반환한다.

    cr3.shopping.naver.com의 nv_mid는
    브랜드스토어 productId와 다른 식별자일 수 있으므로
    여기서는 productId로 취급하지 않는다.
  */
  const match =
    productUrl.match(
      /\/(?:main\/)?products\/(\d+)/,
    );

  return (
    match?.[1] ??
    ""
  );
}

function getNaverShoppingId(
  productUrl: string,
) {
  try {
    const url =
      new URL(productUrl);

    const nvMid =
      url.searchParams
        .get("nv_mid")
        ?.trim();

    if (
      nvMid &&
      /^\d+$/.test(nvMid)
    ) {
      return nvMid;
    }
  } catch {
    // 정규식 fallback 사용
  }

  return (
    productUrl.match(
      /[?&]nv_mid=(\d+)/i,
    )?.[1] ??
    ""
  );
}
function cleanProductName(
  value: string,
) {
  return value
    .replace(
      /\[[^\]]*\]/g,
      " ",
    )
    .replace(
      /\([^)]*\)/g,
      " ",
    )
    .replace(
      /[,/|]+/g,
      " ",
    )
    .replace(
      /\s+/g,
      " ",
    )
    .trim();
}

function getUsefulTokens(
  productName: string,
) {
  const ignored =
    new Set([
      "로봇청소기",
      "청소기",
      "로봇",
      "무선",
      "가정용",
      "자동",
      "단품",
      "화이트",
      "블랙",
      "공식",
      "정품",
      "신제품",
      "인공지능",
      "올인원",
    ]);

  return cleanProductName(
    productName,
  )
    .split(" ")
    .map(
      (token) =>
        token.trim(),
    )
    .filter(
      (token) =>
        token.length >= 2 &&
        !ignored.has(token),
    );
}

function getBrandCandidate(
  productName: string,
) {
  const tokens =
    cleanProductName(
      productName,
    ).split(" ");

  return (
    tokens.find(
      (token) =>
        token.length >= 2 &&
        ![
          "로봇청소기",
          "청소기",
          "로봇",
        ].includes(token),
    ) ?? ""
  );
}

function candidateScore(
  candidate:
    CanonicalCandidate,
  productName: string,
  expectedProductId: string,
) {
  if (
    expectedProductId &&
    candidate.productId ===
      expectedProductId
  ) {
    return 1000;
  }

  const wantedTokens =
    getUsefulTokens(
      productName,
    )
      .map(
        (token) =>
          token.toLowerCase(),
      );

  const title =
    candidate.title
      .toLowerCase();

  let score = 0;

  for (
    const token of
    wantedTokens
  ) {
    if (
      title.includes(token)
    ) {
      score += 10;
    }
  }

  return score;
}

export async function resolveNaverBrandProductUrl(
  productUrl: string,
  productName: string,
) {
  /*
    direct smartstore/brand URL이면
    여기서 상품번호가 바로 잡힌다.

    ader/cr3 같은 중간 URL이면
    productId는 빈 문자열로 시작하고,
    아래 SerpApi 공식상품 검색에서
    실제 productId를 다시 확보한다.
  */
  const inputProductId =
    getProductId(
      productUrl,
    );

  /*
    네이버쇼핑 bridge URL의 nv_mid는
    공식 브랜드스토어 상품번호로 사용하지 않고
    SerpApi 검색 힌트로만 사용한다.
  */
  const naverShoppingId =
    getNaverShoppingId(
      productUrl,
    );

  const cleanedName =
    cleanProductName(
      productName,
    );

  const usefulTokens =
    getUsefulTokens(
      productName,
    );

  const initialBrandCandidate =
    getBrandCandidate(
      productName,
    );

  /*
    이미 학습된 브랜드스토어가 있으면
    SerpApi 검색 전에 즉시 재사용.
  */
  if (
    inputProductId &&
    initialBrandCandidate
  ) {
    const learned =
      await findBrandStoreMapping([
        initialBrandCandidate,
      ]);

    if (
      learned?.brandSite
    ) {
      return {
        success: true as const,

        productId:
          inputProductId,

        brandName:
          learned.brandName ||
          initialBrandCandidate,

        brandSite:
          learned.brandSite,

        canonicalUrl:
          `${learned.brandSite}/products/${inputProductId}`,

        triedQueries:
          [] as string[],
      };
    }
  }

  const modelTokens =
    usefulTokens
      .filter(
        (token) =>
          token !==
          initialBrandCandidate,
      )
      .slice(
        0,
        3,
      );

  const triedQueries:
    string[] = [];

  let brandName = "";

  const canonicalCandidates:
    CanonicalCandidate[] = [];

  const brandSites =
    new Set<string>();

  const tryQuery =
    async (
      query: string,
    ) => {
      const normalized =
        query
          .replace(
            /\s+/g,
            " ",
          )
          .trim();

      if (
        !normalized ||
        triedQueries.includes(
          normalized,
        )
      ) {
        return;
      }

      triedQueries.push(
        normalized,
      );

      const data =
        await searchNaver(
          normalized,
        );

      for (
        const site of
        extractBrandSites(
          data,
        )
      ) {
        brandSites.add(site);
      }

      for (
        const candidate of
        extractCanonicalCandidates(
          data,
        )
      ) {
        if (
          !canonicalCandidates.some(
            (existing) =>
              existing.canonicalUrl ===
              candidate.canonicalUrl,
          )
        ) {
          canonicalCandidates.push(
            candidate,
          );
        }
      }

      if (!brandName) {
        brandName =
          extractBrandName(
            data,
          );
      }
    };

  /*
    상품번호가 이미 있으면
    기존에 성공했던 강한 검색부터 사용.
  */
  if (inputProductId) {
    await tryQuery(
      `${inputProductId} ${cleanedName}`,
    );
  }

  /*
    nv_mid가 있는 네이버쇼핑 bridge URL은
    해당 번호를 검색 힌트로만 사용한다.
    검색 결과에서 실제 brand.naver.com 상품 URL을 찾는다.
  */
  if (
    !inputProductId &&
    naverShoppingId
  ) {
    await tryQuery(
      `${naverShoppingId} ${cleanedName}`,
    );
  }
  /*
    중간 URL 또는 상품번호 미확보 상황에서는
    상품명 자체로 공식 브랜드 상품을 찾는다.
  */
  await tryQuery(
    `${cleanedName} brand.naver.com`,
  );

  if (
    initialBrandCandidate
  ) {
    const shortModel =
      modelTokens
        .slice(0, 2)
        .join(" ");

    await tryQuery(
      `site:brand.naver.com ${initialBrandCandidate} ${shortModel}`,
    );
  }

  const modelLike =
    usefulTokens.filter(
      (token) =>
        /[a-zA-Z]/.test(
          token,
        ) ||
        /\d/.test(
          token,
        ),
    );

  if (
    initialBrandCandidate &&
    modelLike.length > 0
  ) {
    await tryQuery(
      `${initialBrandCandidate} ${modelLike
        .slice(0, 3)
        .join(" ")} 공식`,
    );
  }

  /*
    공식 상품 URL 후보가 확보됐으면
    상품번호 일치 또는 상품명 토큰 일치도가
    가장 높은 후보를 선택한다.
  */
  const rankedCandidates =
    [...canonicalCandidates]
      .map(
        (candidate) => ({
          candidate,
          score:
            candidateScore(
              candidate,
              productName,
              inputProductId,
            ),
        }),
      )
      .sort(
        (a, b) =>
          b.score -
          a.score,
      );

  const best =
    rankedCandidates[0];

  if (
    best &&
    (
      inputProductId
        ? best.candidate
            .productId ===
          inputProductId
        : best.score > 0 ||
          canonicalCandidates.length ===
            1
    )
  ) {
    const resolvedBrandName =
      brandName ||
      initialBrandCandidate;

    try {
      await saveBrandStoreMapping(
        [
          initialBrandCandidate,
          brandName,
          resolvedBrandName,
        ],
        best.candidate
          .brandSite,
        {
          source:
            "serpapi_product_match",
          confidence:
            100,
        },
      );
    } catch (error) {
      console.warn(
        "Brand mapping save warning:",
        error,
      );
    }

    return {
      success: true as const,

      productId:
        best.candidate
          .productId,

      brandName:
        resolvedBrandName,

      brandSite:
        best.candidate
          .brandSite,

      canonicalUrl:
        best.candidate
          .canonicalUrl,

      triedQueries,
    };
  }

  /*
    상품번호는 알고 있지만 검색결과에서
    정확한 상품 URL을 직접 얻지 못한 경우,
    공식 브랜드스토어만 확인되면 기존 방식대로
    brandSite + productId를 조합한다.
  */
  if (
    inputProductId &&
    brandSites.size > 0
  ) {
    const brandSite =
      [...brandSites][0];

    const resolvedBrandName =
      brandName ||
      initialBrandCandidate;

    try {
      await saveBrandStoreMapping(
        [
          initialBrandCandidate,
          brandName,
          resolvedBrandName,
        ],
        brandSite,
        {
          source:
            "serpapi_brand_site",
          confidence:
            95,
        },
      );
    } catch (error) {
      console.warn(
        "Brand mapping save warning:",
        error,
      );
    }

    return {
      success: true as const,

      productId:
        inputProductId,

      brandName:
        resolvedBrandName,

      brandSite,

      canonicalUrl:
        `${brandSite}/products/${inputProductId}`,

      triedQueries,
    };
  }

  if (inputProductId) {
    const learned =
      await findBrandStoreMapping([
        initialBrandCandidate,
        brandName,
      ]);

    if (
      learned?.brandSite
    ) {
      return {
        success: true as const,

        productId:
          inputProductId,

        brandName:
          learned.brandName ||
          initialBrandCandidate ||
          brandName,

        brandSite:
          learned.brandSite,

        canonicalUrl:
          `${learned.brandSite}/products/${inputProductId}`,

        triedQueries,
      };
    }
  }

  /*
    아직 DB에 브랜드스토어가 없는 신규 브랜드라면
    Google에서 네이버 공식 브랜드스토어 자체를 먼저 찾는다.

    예:
    "드리미 네이버 브랜드스토어"
    → https://brand.naver.com/dreame

    발견한 brandSite는 DB에 저장해서
    다음 실행부터 다시 검색하지 않도록 한다.
  */
  let discoveredBrandSite = "";

  const learnedBeforeDiscovery =
    await findBrandStoreMapping([
      initialBrandCandidate,
      brandName,
    ]);

  if (
    !learnedBeforeDiscovery?.brandSite &&
    initialBrandCandidate
  ) {
    try {
      const discoveryQuery =
        `${initialBrandCandidate} 네이버 브랜드스토어`;

      triedQueries.push(
        `[google-brand] ${discoveryQuery}`,
      );

      const apiKey =
        process.env.SERPAPI_API_KEY;

      if (apiKey) {
        const params =
          new URLSearchParams({
            engine: "google",
            q: discoveryQuery,
            api_key: apiKey,
            num: "10",
          });

        const response =
          await fetch(
            "https://serpapi.com/search?" +
              params.toString(),
            {
              cache: "no-store",
            },
          );

        const data =
          (await response.json()) as {
            error?: string;
            organic_results?: {
              title?: string;
              link?: string;
            }[];
          };

        if (
          response.ok &&
          !data.error
        ) {
          const candidates =
            new Set<string>();

          for (
            const result of
            data.organic_results ??
              []
          ) {
            const site =
              normalizeBrandSite(
                result.link,
              );

            if (site) {
              candidates.add(site);
            }
          }

          const serialized =
            JSON.stringify(data);

          const matches =
            serialized.match(
              /https?:\/\/brand\.naver\.com\/[a-zA-Z0-9_-]+/gi,
            ) ?? [];

          for (
            const match of
            matches
          ) {
            const site =
              normalizeBrandSite(
                match,
              );

            if (site) {
              candidates.add(site);
            }
          }

          if (
            candidates.size ===
            1
          ) {
            discoveredBrandSite =
              [...candidates][0];

            try {
              await saveBrandStoreMapping(
                [
                  initialBrandCandidate,
                  brandName,
                ],
                discoveredBrandSite,
                {
                  source:
                    "google_brand_discovery",
                  confidence:
                    95,
                },
              );
            } catch (error) {
              console.warn(
                "Brand mapping save warning:",
                error,
              );
            }
          }
        }
      }
    } catch (error) {
      console.warn(
        "Google brand discovery warning:",
        error,
      );
    }
  }

    Naver SerpApi에서 공식 상품 URL을 찾지 못했을 때
    이미 학습된 브랜드스토어가 있으면
    Google에서 해당 브랜드스토어 내부의 모델명으로만 검색한다.

    예:
    site:brand.naver.com/mova "P70 Pro Ultra"

    특정 브랜드 하드코딩 없이
    DB에 저장된 brandSite를 그대로 사용한다.

      brandName,

                /\d/.test(
                  token,
                )
              ),
          )
          .slice(0, 3);

      if (
        brandSlug &&
        googleModelTokens.length >
          0
      ) {
        const googleQuery =
          `site:brand.naver.com/${brandSlug} "${googleModelTokens.join(
            " ",
          )}"`;

        triedQueries.push(
          `[google] ${googleQuery}`,
        );

        const googleCandidates =
          await searchGoogle(
            googleQuery,
          );

        const sameBrandCandidates =
          googleCandidates.filter(
            (candidate) =>
              candidate.brandSite ===
              googleBrandSite,
          );

        const rankedGoogle =
          sameBrandCandidates
            .map(
              (candidate) => ({
                candidate,

                score:
                  candidateScore(
                    candidate,
                    productName,
                    "",
                  ),
              }),
            )
            .sort(
              (a, b) =>
                b.score -
                a.score,
            );

        const bestGoogle =
          rankedGoogle[0];

        /*
          검색 결과가 하나뿐이면 강하게 신뢰.
          여러 개면 상품명 토큰 점수가 있는 후보만 허용.
        */
        if (
          bestGoogle &&
          (
            sameBrandCandidates.length ===
              1 ||
            bestGoogle.score > 0
          )
        ) {
          try {
            await saveBrandStoreMapping(
              [
                initialBrandCandidate,
                brandName,
                learnedForGoogle?.brandName,
              ],
              bestGoogle.candidate
                .brandSite,
              {
                source:
                  "google_product_match",
                confidence:
                  sameBrandCandidates.length ===
                  1
                    ? 100
                    : 95,
              },
            );
          } catch (error) {
            console.warn(
              "Brand mapping save warning:",
              error,
            );
          }

          return {
            success:
              true as const,

            productId:
              bestGoogle.candidate
                .productId,

            brandName:
              learnedForGoogle?.brandName ||
              initialBrandCandidate ||
              brandName,

            brandSite:
              bestGoogle.candidate
                .brandSite,

            canonicalUrl:
              bestGoogle.candidate
                .canonicalUrl,

            triedQueries,
          };
        }
      }
    } catch (error) {
      console.warn(
        "Google brand product fallback warning:",
        error,
      );
    }
  }
  /*
    Naver SerpApi에서 공식 상품 URL을 찾지 못했을 때
    이미 학습된 브랜드스토어가 있으면

    특정 브랜드 하드코딩 없이
    DB에 저장된 brandSite를 그대로 사용한다.
  */

    await findBrandStoreMapping([
      initialBrandCandidate,
      brandName,
    ]);

  if (

  ) {
    try {
      const brandSlug =
        learnedForGoogle
          .brandSite

          .replace(
            /\/.*$/,
            "",
          )
          .trim();

      const googleModelTokens =
        usefulTokens
          .filter(
            (token) =>
              token !==

              (

      if (

        );

          await searchGoogle(

        const sameBrandCandidates =

          (

              1 ||

          )
        ) {
          try {

              ],
              bestGoogle.candidate
                .brandSite,
              {
                source:
                  "google_product_match",
                confidence:
                  sameBrandCandidates.length ===
                  1
                    ? 100
                    : 95,
              },
            );
          } catch (error) {
            console.warn(
              "Brand mapping save warning:",
              error,
            );
          }

          return {
            success:
              true as const,

            productId:
              bestGoogle.candidate
                .productId,

            brandName:
              learnedForGoogle.brandName ||
              initialBrandCandidate ||

            brandSite:
              bestGoogle.candidate
                .brandSite,

            canonicalUrl:
              bestGoogle.candidate
                .canonicalUrl,

            triedQueries,
