import {
  resolveNaverBrandProductUrl,
} from "../lib/resolveNaverBrandProductUrl";

async function main() {
  const url =
    "https://cr3.shopping.naver.com/v2/bridge/searchGate?nv_mid=59760912197&cat_id=50015541&query=%EB%A1%9C%EB%B4%87%EC%B2%AD%EC%86%8C%EA%B8%B0";

  const name =
    "모바 로봇청소기 P70 Pro Ultra RLP54HE AI센서 자동세척 화이트, 단품";

  const startedAt =
    Date.now();

  const result =
    await resolveNaverBrandProductUrl(
      url,
      name,
    );

  console.log(
    "TIME:",
    Math.round(
      (Date.now() - startedAt) /
        1000,
    ),
    "초",
  );

  console.log(
    JSON.stringify(
      result,
      null,
      2,
    ),
  );
}

main().catch(
  (error) => {
    console.error(error);
    process.exit(1);
  },
);
