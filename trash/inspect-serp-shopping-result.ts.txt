const apiKey =
  process.env.SERPAPI_API_KEY;

if (!apiKey) {
  throw new Error(
    "SERPAPI_API_KEY가 없습니다.",
  );
}

async function main() {
  const params =
    new URLSearchParams({
      engine: "naver",
      query: "로봇청소기 인기",
      where: "nexearch",
      output: "json",
      api_key: apiKey,
    });

  const response =
    await fetch(
      "https://serpapi.com/search?" +
        params.toString(),
    );

  const data: any =
    await response.json();

  if (!response.ok) {
    console.error(data);
    process.exit(1);
  }

  const results =
    Array.isArray(
      data.shopping_results,
    )
      ? data.shopping_results
      : [];

  console.log(
    "shopping_results:",
    results.length,
  );

  for (
    let i = 0;
    i < Math.min(
      results.length,
      5,
    );
    i++
  ) {
    console.log(
      `\n===== RESULT ${i + 1} =====`,
    );

    console.log(
      JSON.stringify(
        results[i],
        null,
        2,
      ),
    );
  }
}

main().catch(
  (error) => {
    console.error(error);
    process.exit(1);
  },
);
