import {
  collectNaverProduct,
} from "../lib/brightDataProduct";

async function main() {
  const url =
    "https://search.shopping.naver.com/catalog/59760912197";

  console.log(
    "Bright Data catalog URL 테스트 시작",
  );
  console.log("URL:", url);

  const startedAt =
    Date.now();

  try {
    const result =
      await collectNaverProduct(
        url,
      );

    console.log("");
    console.log(
      "===== 성공 =====",
    );

    console.log(
      "시간:",
      Math.round(
        (Date.now() -
          startedAt) /
          1000,
      ),
      "초",
    );

    console.log(
      JSON.stringify(
        {
          url:
            result.url,
          productId:
            result.productId,
          title:
            result.title,
          finalPrice:
            result.finalPrice,
          totalReviews:
            result.totalReviews,
          brand:
            result.brand,
          modelName:
            result.modelName,
        },
        null,
        2,
      ),
    );
  } catch (error) {
    console.log("");
    console.log(
      "===== 실패 =====",
    );

    console.log(
      "시간:",
      Math.round(
        (Date.now() -
          startedAt) /
          1000,
      ),
      "초",
    );

    console.error(error);
  }
}

main();
