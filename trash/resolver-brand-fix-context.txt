    add(match);
  }

  return [
    ...candidates.values(),
  ];
}

function extractBrandName(
  data: SerpApiResult,
) {
  for (
    const result of
    data.web_results ?? []
  ) {
    const combined =
      [
        result.title,
        result.snippet,
      ]
        .filter(Boolean)
        .join(" ");

    const match =
      combined.match(
        /브랜드\s*[:：]\s*([가-힣A-Za-z0-9._+-]+)/,
      );

    if (match?.[1]) {
      return match[1].trim();
    }
  }

  return "";
}
