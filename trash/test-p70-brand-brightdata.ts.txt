import {
  collectNaverProduct,
} from "../lib/brightDataProduct";

async function run() {
  const url =
    "https://brand.naver.com/mova/products/13280227814";

  console.log(
    "TEST:",
    url,
  );

  const started =
    Date.now();

  try {
    const result =
      await collectNaverProduct(
        url,
      );

    console.log(
      "TIME:",
      Math.round(
        (Date.now() - started) /
          1000,
      ),
      "초",
    );

    console.log(
      JSON.stringify(
        {
          productId:
            result.productId,

          title:
            result.title,

          finalPrice:
            result.finalPrice,

          totalReviews:
            result.totalReviews,

          averageRating:
            result.averageRating,

          brand:
            result.brand,

          manufacturer:
            result.manufacturer,

          modelName:
            result.modelName,

          categoryName:
            result.categoryName,
        },
        null,
        2,
      ),
    );
  } catch (error) {
    console.log(
      "ERROR:",
      error instanceof Error
        ? error.message
        : error,
    );
  }
}

run();
