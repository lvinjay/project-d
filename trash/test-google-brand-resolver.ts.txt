const apiKey =
  process.env.SERPAPI_API_KEY;

if (!apiKey) {
  throw new Error(
    "SERPAPI_API_KEY가 없습니다.",
  );
}

async function search(
  query: string,
) {
  const params =
    new URLSearchParams({
      engine: "google",
      q: query,
      api_key: apiKey,
      num: "10",
    });

  const response =
    await fetch(
      "https://serpapi.com/search?" +
        params.toString(),
    );

  const data: any =
    await response.json();

  console.log("");
  console.log(
    "QUERY:",
    query,
  );

  if (!response.ok || data.error) {
    console.log(
      "ERROR:",
      data.error ??
        response.status,
    );
    return;
  }

  const text =
    JSON.stringify(data);

  const matches =
    text.match(
      /https?:\/\/brand\.naver\.com\/mova\/products\/\d+/gi,
    ) ?? [];

  console.log(
    "MATCHES:",
    [
      ...new Set(matches),
    ],
  );

  const organic =
    Array.isArray(
      data.organic_results,
    )
      ? data.organic_results
          .slice(0, 5)
          .map(
            (item: any) => ({
              title:
                item.title,
              link:
                item.link,
            }),
          )
      : [];

  console.log(
    "TOP RESULTS:",
    JSON.stringify(
      organic,
      null,
      2,
    ),
  );
}

async function main() {
  await search(
    'site:brand.naver.com/mova "P70 Pro Ultra"',
  );

  await search(
    'site:brand.naver.com/mova "RLP54HE"',
  );

  await search(
    '"모바 P70 Pro Ultra" "brand.naver.com"',
  );
}

main().catch(
  (error) => {
    console.error(error);
    process.exit(1);
  },
);
