type SerpApiResult = {
  ads_results?: {
    site?: string;
    title?: string;
    sub_title?: string;
    description?: string;
  }[];

  web_results?: {
    title?: string;
    link?: string;
    snippet?: string;
  }[];

  shopping_results?: {
    title?: string;
    link?: string;
    stores?: string;
    price?: number | string;
    rating?: number | string;
    reviews?: number | string;
  }[];
};

type SerpApiResponse =
  SerpApiResult & {
    error?: string;
  };

type CanonicalCandidate = {
  productId: string;
  brandSite: string;
  canonicalUrl: string;
  title: string;
};

function normalizeBrandSite(
  value: unknown,
) {
  const text =
    String(value ?? "")
      .trim();

  const match =
    text.match(
      /(?:https?:\/\/)?brand\.naver\.com\/([a-zA-Z0-9_-]+)/i,
    );

  if (!match) {
    return "";
  }

  return (
    "https://brand.naver.com/" +
    match[1]
  );
}

function normalizeCanonicalProductUrl(
  value: unknown,
) {
  const text =
    String(value ?? "")
      .trim();

  const match =
    text.match(
      /https?:\/\/brand\.naver\.com\/([a-zA-Z0-9_-]+)\/products\/(\d+)/i,
    );

  if (!match) {
    return null;
  }

  const brandSite =
    `https://brand.naver.com/${match[1]}`;

  const productId =
    match[2];

  return {
    productId,
    brandSite,
    canonicalUrl:
      `${brandSite}/products/${productId}`,
  };
}

async function searchNaver(
  query: string,
): Promise<SerpApiResponse> {
  const apiKey =
    process.env.SERPAPI_API_KEY;

  if (!apiKey) {
    throw new Error(
      "SERPAPI_API_KEY가 설정되지 않았습니다.",
    );
  }

  const params =
    new URLSearchParams({
      engine: "naver",
      query,
      where: "nexearch",
      output: "json",
      api_key: apiKey,
    });

  const response =
    await fetch(
      "https://serpapi.com/search?" +
        params.toString(),
      {
        cache: "no-store",
      },
    );

  const data =
    (await response.json()) as
      SerpApiResponse;

  if (
    !response.ok ||
    data.error
  ) {
    throw new Error(
      data.error ??
        `SerpApi 검색 실패 (${response.status})`,
    );
  }

  return data;
}

function extractBrandSites(
  data: SerpApiResult,
) {
  const sites =
    new Set<string>();

  const add = (
    value: unknown,
  ) => {
    const site =
      normalizeBrandSite(
        value,
      );

    if (site) {
      sites.add(site);
    }
  };

  for (
    const ad of
    data.ads_results ?? []
  ) {
    add(ad.site);
  }

  for (
    const result of
    data.web_results ?? []
  ) {
    add(result.link);
  }

  for (
    const result of
    data.shopping_results ?? []
  ) {
    add(result.link);
  }

  const serialized =
    JSON.stringify(data);

  const matches =
    serialized.match(
      /brand\.naver\.com\/[a-zA-Z0-9_-]+/gi,
    ) ?? [];

  for (
    const match of
    matches
  ) {
    add(match);
  }

  return [...sites];
}

function extractCanonicalCandidates(
  data: SerpApiResult,
) {
  const candidates =
    new Map<
      string,
      CanonicalCandidate
    >();

  const add = (
    value: unknown,
    title = "",
  ) => {
    const normalized =
      normalizeCanonicalProductUrl(
        value,
      );

    if (!normalized) {
      return;
    }

    const key =
      normalized.canonicalUrl;

    if (
      !candidates.has(key)
    ) {
      candidates.set(
        key,
        {
          ...normalized,
          title,
        },
      );
    }
  };

  for (
    const result of
    data.web_results ?? []
  ) {
    add(
      result.link,
      String(
        result.title ?? "",
      ),
    );
  }

  for (
    const result of
    data.shopping_results ?? []
  ) {
    add(
      result.link,
      String(
        result.title ?? "",
      ),
    );
  }

  /*
    구조화 필드 밖에 공식 상품 URL이
    들어오는 경우도 대응한다.
  */
  const serialized =
    JSON.stringify(data);

  const matches =
    serialized.match(
      /https?:\/\/brand\.naver\.com\/[a-zA-Z0-9_-]+\/products\/\d+/gi,
    ) ?? [];

  for (
    const match of
    matches
  ) {
    add(match);
  }

  return [
    ...candidates.values(),
  ];
}

function extractBrandName(
  data: SerpApiResult,
) {
  for (
    const result of
    data.web_results ?? []
  ) {
    const combined =
      [
        result.title,
        result.snippet,
      ]
        .filter(Boolean)
        .join(" ");

    const match =
      combined.match(
        /브랜드\s*[:：]\s*([가-힣A-Za-z0-9._+-]+)/,
      );

    if (match?.[1]) {
      return match[1].trim();
    }
  }

  return "";
}

function getProductId(
  productUrl: string,
) {
  const match =
    productUrl.match(
      /\/(?:main\/)?products\/(\d+)/,
    );

  return (
    match?.[1] ??
    ""
  );
}

function cleanProductName(
  value: string,
) {
  return value
    .replace(
      /\[[^\]]*\]/g,
      " ",
    )
    .replace(
      /\([^)]*\)/g,
      " ",
    )
    .replace(
      /[,/|]+/g,
      " ",
    )
    .replace(
      /\s+/g,
      " ",
    )
    .trim();
}

function getUsefulTokens(
  productName: string,
) {
  const ignored =
    new Set([
      "로봇청소기",
      "청소기",
      "로봇",
      "무선",
      "가정용",
      "자동",
      "단품",
      "화이트",
      "블랙",
      "공식",
      "정품",
      "신제품",
      "인공지능",
      "올인원",
    ]);

  return cleanProductName(
    productName,
  )
    .split(" ")
    .map(
      (token) =>
        token.trim(),
    )
    .filter(
      (token) =>
        token.length >= 2 &&
        !ignored.has(token),
    );
}

function getBrandCandidate(
  productName: string,
) {
  const tokens =
    cleanProductName(
      productName,
    ).split(" ");

  return (
    tokens.find(
      (token) =>
        token.length >= 2 &&
        ![
          "로봇청소기",
          "청소기",
          "로봇",
        ].includes(token),
    ) ?? ""
  );
}

function candidateScore(
  candidate:
    CanonicalCandidate,
  productName: string,
  expectedProductId: string,
) {
  if (
    expectedProductId &&
    candidate.productId ===
      expectedProductId
  ) {
    return 1000;
  }

  const wantedTokens =
    getUsefulTokens(
      productName,
    )
      .map(
        (token) =>
          token.toLowerCase(),
      );

  const title =
    candidate.title
      .toLowerCase();

  let score = 0;

  for (
    const token of
    wantedTokens
  ) {
    if (
      title.includes(token)
    ) {
      score += 10;
    }
  }

  return score;
}

export async function resolveNaverBrandProductUrl(
  productUrl: string,
  productName: string,
) {
  /*
    direct smartstore/brand URL이면
    여기서 상품번호가 바로 잡힌다.

    ader/cr3 같은 중간 URL이면
    productId는 빈 문자열로 시작하고,
    아래 SerpApi 공식상품 검색에서
    실제 productId를 다시 확보한다.
  */
  const inputProductId =
    getProductId(
      productUrl,
    );

  const cleanedName =
    cleanProductName(
      productName,
    );

  const usefulTokens =
    getUsefulTokens(
      productName,
    );

  const initialBrandCandidate =
    getBrandCandidate(
      productName,
    );

  const modelTokens =
    usefulTokens
      .filter(
        (token) =>
          token !==
          initialBrandCandidate,
      )
      .slice(
        0,
        3,
      );

  const triedQueries:
    string[] = [];

  let brandName = "";

  const canonicalCandidates:
    CanonicalCandidate[] = [];

  const brandSites =
    new Set<string>();

  const tryQuery =
    async (
      query: string,
    ) => {
      const normalized =
        query
          .replace(
            /\s+/g,
            " ",
          )
          .trim();

      if (
        !normalized ||
        triedQueries.includes(
          normalized,
        )
      ) {
        return;
      }

      triedQueries.push(
        normalized,
      );

      const data =
        await searchNaver(
          normalized,
        );

      for (
        const site of
        extractBrandSites(
          data,
        )
      ) {
        brandSites.add(site);
      }

      for (
        const candidate of
        extractCanonicalCandidates(
          data,
        )
      ) {
        if (
          !canonicalCandidates.some(
            (existing) =>
              existing.canonicalUrl ===
              candidate.canonicalUrl,
          )
        ) {
          canonicalCandidates.push(
            candidate,
          );
        }
      }

      if (!brandName) {
        brandName =
          extractBrandName(
            data,
          );
      }
    };

  /*
    상품번호가 이미 있으면
    기존에 성공했던 강한 검색부터 사용.
  */
  if (inputProductId) {
    await tryQuery(
      `${inputProductId} ${cleanedName}`,
    );
  }

  /*
    중간 URL 또는 상품번호 미확보 상황에서는
    상품명 자체로 공식 브랜드 상품을 찾는다.
  */
  await tryQuery(
    `${cleanedName} brand.naver.com`,
  );

  if (
    initialBrandCandidate
  ) {
    const shortModel =
      modelTokens
        .slice(0, 2)
        .join(" ");

    await tryQuery(
      `site:brand.naver.com ${initialBrandCandidate} ${shortModel}`,
    );
  }

  const modelLike =
    usefulTokens.filter(
      (token) =>
        /[a-zA-Z]/.test(
          token,
        ) ||
        /\d/.test(
          token,
        ),
    );

  if (
    initialBrandCandidate &&
    modelLike.length > 0
  ) {
    await tryQuery(
      `${initialBrandCandidate} ${modelLike
        .slice(0, 3)
        .join(" ")} 공식`,
    );
  }

  /*
    공식 상품 URL 후보가 확보됐으면
    상품번호 일치 또는 상품명 토큰 일치도가
    가장 높은 후보를 선택한다.
  */
  const rankedCandidates =
    [...canonicalCandidates]
      .map(
        (candidate) => ({
          candidate,
          score:
            candidateScore(
              candidate,
              productName,
              inputProductId,
            ),
        }),
      )
      .sort(
        (a, b) =>
          b.score -
          a.score,
      );

  const best =
    rankedCandidates[0];

  if (
    best &&
    (
      inputProductId
        ? best.candidate
            .productId ===
          inputProductId
        : best.score > 0 ||
          canonicalCandidates.length ===
            1
    )
  ) {
    return {
      success: true as const,

      productId:
        best.candidate
          .productId,

      brandName:
        brandName ||
        initialBrandCandidate,

      brandSite:
        best.candidate
          .brandSite,

      canonicalUrl:
        best.candidate
          .canonicalUrl,

      triedQueries,
    };
  }

  /*
    상품번호는 알고 있지만 검색결과에서
    정확한 상품 URL을 직접 얻지 못한 경우,
    공식 브랜드스토어만 확인되면 기존 방식대로
    brandSite + productId를 조합한다.
  */
  if (
    inputProductId &&
    brandSites.size > 0
  ) {
    const brandSite =
      [...brandSites][0];

    return {
      success: true as const,

      productId:
        inputProductId,

      brandName:
        brandName ||
        initialBrandCandidate,

      brandSite,

      canonicalUrl:
        `${brandSite}/products/${inputProductId}`,

      triedQueries,
    };
  }

  /*
    마지막 fallback:

    검색결과에서 brand.naver.com 주소는 못 찾았지만
    영문 브랜드명을 확보한 경우
    네이버 브랜드스토어 slug 후보로 사용한다.

    예:
    MOVA -> mova
    https://brand.naver.com/mova/products/13280227814

    한글 브랜드명은 slug를 안전하게 추정할 수 없으므로
    이 fallback을 사용하지 않는다.
  */
  const fallbackBrandName =
    (
      brandName ||
      initialBrandCandidate
    ).trim();

  const brandSlugAliases:
    Record<string, string> = {
      "모바": "mova",
      "MOVA": "mova",
      "에코백스": "ecovacs",
      "ECOVACS": "ecovacs",
      "로보락": "roborock",
      "Roborock": "roborock",
      "에브리봇": "everybot",
    };

  const fallbackBrandSlug =
    brandSlugAliases[
      fallbackBrandName
    ] ??
    (
      /^[A-Za-z0-9_-]+$/.test(
        fallbackBrandName,
      )
        ? fallbackBrandName.toLowerCase()
        : ""
    );

  if (
    inputProductId &&
    fallbackBrandSlug
  ) {
    const brandSite =
      `https://brand.naver.com/${fallbackBrandSlug}`;

    return {
      success: true as const,

      productId:
        inputProductId,

      brandName:
        fallbackBrandName,

      brandSite,

      canonicalUrl:
        `${brandSite}/products/${inputProductId}`,

      triedQueries,
    };
  }

  return {
    success: false as const,

    productId:
      inputProductId,

    brandName:
      brandName ||
      initialBrandCandidate,

    brandSite:
      [...brandSites][0] ??
      "",

    canonicalUrl: "",

    triedQueries,

    reason:
      inputProductId
        ? "brand.naver.com 공식 상품 URL을 찾지 못했습니다."
        : "중간 네이버 URL에서 공식 상품번호와 brand.naver.com 상품 URL을 확인하지 못했습니다.",
  };
}
